\section*{Soluciones}

\begin{enumerate}
    \item \textbf{Si $P(x) = x^2 - x + 2$ y $Q(x) = x^3 - x^2 + 1$, calcula:}
    \begin{enumerate}[label=\textbf{\alph*)}]
        \item $P(1) + P(-1) = (1 - 1 + 2) + (1 + 1 + 2) = 2 + 4 = 6$
        \item $P(0) + Q(-1) = 2 + (-1 - 1 + 1) = 2 + (-1) = 1$
    \end{enumerate}

    \vspace{0.5cm}

    \item \textbf{¿Cuánto tiene que valer $a$ para que $P(a) = 0$ si $P(x) = 2x^2 - 3x + 1$?}

    Son las soluciones de la ecuación $2x^2 - 3x + 1 = 0 \longrightarrow a = 1$ y $a = \dfrac{1}{2}$

    \vspace{0.5cm}

    \item \textbf{Realiza las siguientes divisiones de polinomios. Comprueba, en cada una de ellas, el resultado que obtienes.}
        \begin{enumerate}[label=\textbf{\alph*)}]
            \item $(2x^3 - 3x^2 - 5x - 5) : (x^2 - 2x - 1) \longrightarrow \begin{cases} \text{Cociente } C(x) = 2x + 1 \\ \text{Resto } R(x) = -x - 4 \end{cases}$

            \vspace{0.3cm}

            \item $(2x^3 - 3x^2 + 4x - 3) : (x^2 - 1) \longrightarrow \begin{cases} \text{Cociente } C(x) = 2x - 3 \\ \text{Resto } R(x) = 6x - 6 \end{cases}$

            \vspace{0.3cm}

            \item $(x^4 + 1) : (x^2 + 1) \longrightarrow \begin{cases} \text{Cociente } C(x) = x^2 - 1 \\ \text{Resto } R(x) = 2 \end{cases}$

            \vspace{0.3cm}

            \item $(x^5 + 2x^3 - 1) : (x^2 - 3) \longrightarrow \begin{cases} \text{Cociente } C(x) = x^3 + 5x \\ \text{Resto } R(x) = 15x - 1 \end{cases}$
        \end{enumerate}

    \vspace{0.5cm}

    \item \textbf{El dividendo de una división de polinomios es $P(x) = x^5 - 2x^3 - x^2$, el cociente es $C(x) = x^2 - 2$ y el resto es $R(x) = -2$. ¿Cuál es el divisor?}

        \vspace{0.3cm}
        Aplicando la propiedad de la división: $P(x) = Q(x) \cdot C(x) + R(x)$
        \[
        x^5 - 2x^3 - x^2 = Q(x) \cdot (x^2 - 2) - 2 \longrightarrow x^5 - 2x^3 - x^2 + 2 = Q(x) \cdot (x^2 - 2)
        \]
        \[
        Q(x) = (x^5 - 2x^3 - x^2 + 2) : (x^2 - 2) = x^3 - 1
        \]

    \vspace{0.5cm}

    \item \textbf{Determina el cociente y el resto, aplicando la regla de Ruffini.}
        \begin{enumerate}[label=\textbf{\alph*)}]
            \item $(x^3 - x^2 + x - 3) : (x - 1)$
            \begin{center}
            \begin{tabular}{r|rrrr}
              & 1 & $-1$ & 1 & $-3$ \\
            1 &   & 1 & 0 & 1 \\ \hline
              & 1 & 0 & 1 & \multicolumn{1}{|r}{$-2$}
            \end{tabular}
            \quad $\longrightarrow \quad C(x) = x^2 + 1; \quad R(x) = -2$
            \end{center}

            \vspace{0.4cm}

            \item $(x^4 - x^3 - x + 9) : (x - 2)$
            \begin{center}
            \begin{tabular}{r|rrrrr}
              & 1 & $-1$ & 0 & $-1$ & 9 \\
            2 &   & 2 & 2 & 4 & 6 \\ \hline
              & 1 & 1 & 2 & 3 & \multicolumn{1}{|r}{15}
            \end{tabular}
            \quad $\longrightarrow \quad C(x) = x^3 + x^2 + 2x + 3; \quad R(x) = 15$
            \end{center}

            \vspace{0.4cm}

            \item $(x^4 + x^2 - 10) : (x - 5)$
            \begin{center}
            \begin{tabular}{r|rrrrr}
              & 1 & 0 & 1 & 0 & $-10$ \\
            5 &   & 5 & 25 & 130 & 650 \\ \hline
              & 1 & 5 & 26 & 130 & \multicolumn{1}{|r}{640}
            \end{tabular}
            
            \smallskip
            $C(x) = x^3 + 5x^2 + 26x + 130; \quad R(x) = 640$
            \end{center}

            \vspace{0.4cm}

            \item $(x^5 - 2x^3 + x - 7) : (x + 3)$
            \begin{center}
            \begin{tabular}{r|rrrrrr}
              & 1 & 0 & $-2$ & 0 & 1 & $-7$ \\
            $-3$ &   & $-3$ & 9 & $-21$ & 63 & $-192$ \\ \hline
              & 1 & $-3$ & 7 & $-21$ & 64 & \multicolumn{1}{|r}{$-199$}
            \end{tabular}
            
            \smallskip
            $C(x) = x^4 - 3x^3 + 7x^2 - 21x + 64; \quad R(x) = -199$
            \end{center}

            \vspace{0.4cm}

            \item $(x^7 + x^4 - 7x^2) : (x + 4)$
            \begin{center}
            \small
            \begin{tabular}{r|rrrrrrrr}
              & 1 & 0 & 0 & 1 & 0 & $-7$ & 0 & 0 \\
            $-4$ &   & $-4$ & 16 & $-64$ & 252 & $-1008$ & 4060 & $-16240$ \\ \hline
              & 1 & $-4$ & 16 & $-63$ & 252 & $-1015$ & 4060 & \multicolumn{1}{|r}{$-16240$}
            \end{tabular}
            
            \smallskip
            $C(x) = x^6 - 4x^5 + 16x^4 - 63x^3 + 252x^2 - 1015x + 4060; \quad R(x) = -16240$
            \end{center}
        \end{enumerate}

    \vspace{0.5cm}

    \item \textbf{Calcula el valor de $m$ para que la división sea exacta.}
    \[
    (x^5 - 2x^3 - 8x^2 + mx + 3) : (x - 3)
    \]
    \begin{center}
    \begin{tabular}{r|rrrrrr}
      & 1 & 0 & $-2$ & $-8$ & $m$ & 3 \\
    3 &   & 3 & 9 & 21 & 39 & $117 + 3m$ \\ \hline
      & 1 & 3 & 7 & 13 & $39 + m$ & \multicolumn{1}{|r}{$120 + 3m$}
    \end{tabular}
    \end{center}

    \[
    120 + 3m = 0 \longrightarrow 3m = -120 \longrightarrow m = -40
    \]

    \vspace{0.5cm}

    \item \textbf{Simplifica estas fracciones algebraicas.}
        \begin{enumerate}[label=\textbf{\alph*)}]
            \item $\dfrac{2x - 2}{2x - 6} = \dfrac{2(x - 1)}{2(x - 3)} = \dfrac{x - 1}{x - 3}$
            
            \vspace{0.3cm}
            
            \item $\dfrac{x^2 - 1}{x^2 - 4x + 3} = \dfrac{(x + 1)(x - 1)}{(x - 1)(x - 3)} = \dfrac{x + 1}{x - 3}$
            
            \vspace{0.3cm}
            
            \item $\dfrac{x - 1}{x^2 - 1} = \dfrac{x - 1}{(x + 1)(x - 1)} = \dfrac{1}{x + 1}$
            
            \vspace{0.3cm}
            
            \item $\dfrac{2x^3 + 4x^2 + 2x}{6x^3 - 6x} = \dfrac{2x(x + 1)^2}{6x(x + 1)(x - 1)} = \dfrac{x + 1}{3(x - 1)}$
            
            \vspace{0.3cm}
            
            \item $\dfrac{x^3 + 3x^2 - 4}{x^2 - 5x + 4} = \dfrac{(x - 1)(x + 2)^2}{(x - 4)(x - 1)} = \dfrac{(x + 2)^2}{x - 4}$
            
            \vspace{0.3cm}
            
            \item $\dfrac{x^2 - 1}{x^2 - 2x + 1} = \dfrac{(x - 1)(x + 1)}{(x - 1)^2} = \dfrac{x + 1}{x - 1}$
        \end{enumerate}

    \vspace{0.5cm}

    \item \textbf{Pasar a álgebra las siguientes expresiones:}
        \begin{enumerate}[label=\textbf{\alph*)}]
            \item Un número “x” es 6 unidades mayor que la mitad de otro “y”.
            \[
            x = \frac{y}{2} + 6
            \]

            \item La suma de las edades que tendrán una madre y una hija dentro de 10 años ($m$ madre, $h$ hija).
            \[
            (m + 10) + (h + 10) = m + h + 20
            \]

            \item El perímetro de un rectángulo es tres veces su altura ($b$ base, $h$ altura).
            \[
            2b + 2h = 3h \quad \text{o bien} \quad 2b = h
            \]

            \item Dentro de 10 años la edad de mi padre será igual a la de mi madre actualmente ($p$ padre, $m$ madre).
            \[
            p + 10 = m
            \]

            \item La diferencia de los inversos de dos números ($x$ e $y$) disminuida en 10 unidades.
            \[
            \frac{1}{x} - \frac{1}{y} - 10
            \]

            \item El cociente de dos números ($x$ e $y$) es 8.
            \[
            \frac{x}{y} = 8
            \]

            \item El cociente de dos números es 6 y el resto 4 ($D$ dividendo, $d$ divisor).
            \[
            D = 6d + 4 \quad \text{o bien} \quad \frac{D}{d} = 6 + \frac{4}{d}
            \]

            \item La suma de los cuadrados de dos números consecutivos ($x$ y $x+1$).
            \[
            x^2 + (x + 1)^2
            \]

            \item El producto de dos números es 143 veces su cociente.
            \[
            x \cdot y = 143 \cdot \frac{x}{y}
            \]
        \end{enumerate}

\end{enumerate}
